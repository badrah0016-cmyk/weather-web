function toNumber(value: Num): number | null {
    if (value === null || value === undefined || value === "") return null;

    const result = Number(value);

    return Number.isFinite(result) ? result : null;
}

function parseDate(value: string | null | undefined): Date | null {
    if (!value) return null;

    const text = String(value).trim();
    if (!text) return null;

    const isoDate = new Date(text);
    if (!Number.isNaN(isoDate.getTime())) return isoDate;

    const match = text.match(
        /^(\d{4})-(\d{2})-(\d{2})[ T](\d{1,2}):(\d{2})(?::(\d{2}))?/
    );

    if (match) {
        return new Date(
            Number(match[1]),
            Number(match[2]) - 1,
            Number(match[3]),
            Number(match[4]),
            Number(match[5]),
            Number(match[6] || 0)
        );
    }

    return null;
}

function getHourlyDateTime(item: Weather): Date | null {
    const observed =
        item.ajiglasan_ognoo ??
        item.hemjsen_ognoo ??
        item.hadgalsan_ognoo;

    if (item.tsag && String(item.tsag).includes("-")) {
        const date = parseDate(String(item.tsag));
        if (date) return date;
    }

    if (item.tsag && observed) {
        const timeMatch = String(item.tsag).match(
            /^(\d{1,2}):(\d{2})(?::(\d{2}))?$/
        );
        const observedDate = parseDate(String(observed));

        if (timeMatch && observedDate) {
            return new Date(
                observedDate.getFullYear(),
                observedDate.getMonth(),
                observedDate.getDate(),
                Number(timeMatch[1]),
                Number(timeMatch[2]),
                Number(timeMatch[3] || 0)
            );
        }
    }

    return parseDate(String(observed || ""));
}

function getClosestHourly(list: Weather[]): Weather | null {
    if (!list.length) return null;

    const now = Date.now();
    let closest: Weather | null = null;
    let difference = Infinity;

    for (const item of list) {
        const date = getHourlyDateTime(item);
        if (!date) continue;

        const diff = Math.abs(date.getTime() - now);

        if (diff < difference) {
            difference = diff;
            closest = item;
        }
    }

    return closest && difference <= 2 * 60 * 60 * 1000 ? closest : null;
}

function extractArray<T>(data: unknown): T[] {
    if (Array.isArray(data)) return data as T[];

    if (data && typeof data === "object") {
        const obj = data as Record<string, unknown>;

        for (const key of ["data", "items", "result", "forecast", "hourly"]) {
            if (Array.isArray(obj[key])) return obj[key] as T[];
        }
    }

    return [];
}

function firstOf<T>(data: unknown): T | null {
    const list = extractArray<T>(data);

    if (list.length > 0) return list[0];

    return data && typeof data === "object" ? (data as T) : null;
}

async function getJson(url: string): Promise<unknown> {
    const response = await fetch(url, { cache: "no-store" });

    if (!response.ok) {
        throw new Error(`HTTP ${response.status}: ${url}`);
    }

    return response.json();
}

const MINUS = "\u2212";
const EMPTY = "--";

// Тоо дэх хасах тэмдгийг жинхэнэ "−" болгоно
function tempNumber(value: number | null): string {
    if (value === null) return EMPTY;

    const r = Math.round(value);

    return `${r < 0 ? MINUS : ""}${Math.abs(r)}`;
}

function formatTemp(value: number | null): string {
    return value === null ? EMPTY : `${tempNumber(value)}°`;
}

function formatValue(
    value: number | null,
    suffix = "",
    digits = 0
): string {
    return value === null ? EMPTY : `${value.toFixed(digits)}${suffix}`;
}

// Хэмжсэн цагийг HH:mm хэлбэрээр
function clockText(value: string | null | undefined): string | null {
    if (!value) return null;

    const text = String(value).trim();

    if (/[zZ]$|[+-]\d{2}:?\d{2}$/.test(text)) {
        const date = new Date(text);

        if (!Number.isNaN(date.getTime())) {
            return new Intl.DateTimeFormat("mn-MN", {
                hour: "2-digit",
                minute: "2-digit",
                hour12: false,
                timeZone: "Asia/Ulaanbaatar",
            }).format(date);
        }
    }

    const match = text.match(/(\d{1,2}):(\d{2})/);

    return match ? `${match[1].padStart(2, "0")}:${match[2]}` : null;
}

function getWindDirection(value: Num): string | null {
    const degree = toNumber(value);

    if (degree === null) return null;

    const directions = [
        "хойд",
        "зүүн хойд",
        "зүүн",
        "зүүн өмнөд",
        "өмнөд",
        "баруун өмнөд",
        "баруун",
        "баруун хойд",
    ];

    return directions[Math.round(degree / 45) % 8];
}

function getWeekday(forecast: Forecast, index: number): string {
    if (forecast.garig) return forecast.garig;

    if (forecast.ognoo) {
        const date = parseDate(forecast.ognoo);

        if (date) {
            return new Intl.DateTimeFormat("mn-MN", {
                weekday: "long",
                timeZone: "Asia/Ulaanbaatar",
            }).format(date);
        }
    }

    const days = ["Ням", "Даваа", "Мягмар", "Лхагва", "Пүрэв", "Баасан", "Бямба"];

    return days[index % 7];
}

    return (
        AQI_BANDS.find((band) => aqi <= band.max)?.color ??

"use client";

import { useCallback, useEffect, useMemo, useState } from "react";

import {
  MapContainer,
  TileLayer,
  GeoJSON,
  Marker,
  Popup,
  useMap,
  useMapEvents,
} from "react-leaflet";

import L from "leaflet";
import type { GeoJsonObject } from "geojson";

import "leaflet/dist/leaflet.css";

interface MongoliaMapProps {
  geoJson?: GeoJsonObject;
  selectedId?: number | string | null;
  onSelect?: (id: number | string) => void;
}

interface Station {
  stants_dugaar: number | string;
  ner: string;
  ner_en?: string;
  aimag?: string;
  urgurug: number;
  urtrag: number;
}

interface CurrentWeather {
  temperatur_c?: number | string | null;
  medregdeh_c?: number | string | null;
  chigleg_huvi?: number | string | null;
  daralt_hpa?: number | string | null;
  salhiny_hurd_ms?: number | string | null;
  salhiny_chiglel?: string | null;
  tsag_agaar_nohtsol?: string | null;
  hur_tunadas_mm?: number | string | null;
  ajiglasan_ognoo?: string | null;
  hemjsen_ognoo?: string | null;
  hadgalsan_ognoo?: string | null;
}

interface StationWeather extends Station {
  weather: CurrentWeather | null;
}

const API_BASE_URL = "/backend";

const MONGOLIA_CENTER: [number, number] = [46.8625, 103.8467];

function getNumber(value: unknown): number | null {
  if (value === null || value === undefined || value === "") {
    return null;
  }

  const numberValue = Number(value);

  return Number.isFinite(numberValue) ? numberValue : null;
}

function getArray(value: unknown): unknown[] {
  if (Array.isArray(value)) {
    return value;
  }

  if (value && typeof value === "object") {
    const object = value as Record<string, unknown>;

    for (const key of ["data", "items", "results", "stations"]) {
      if (Array.isArray(object[key])) {
        return object[key] as unknown[];
      }
    }
  }

  return [];
}

function getWeatherObject(value: unknown): CurrentWeather | null {
  if (!value || typeof value !== "object") {
    return null;
  }

  const object = value as Record<string, unknown>;

  for (const key of ["data", "weather", "result", "current"]) {
    const nested = object[key];

    if (nested && typeof nested === "object" && !Array.isArray(nested)) {
      return nested as CurrentWeather;
    }

    if (Array.isArray(nested) && nested.length > 0) {
      const first = nested[0];

      if (first && typeof first === "object") {
        return first as CurrentWeather;
      }
    }
  }

  return object as CurrentWeather;
}

function formatTemperature(value: unknown): string {
  const temperature = getNumber(value);

  if (temperature === null) {
    return "--°";
  }

  return `${Math.round(temperature)}°`;
}

function getTemperatureColor(value: unknown): string {
  const temperature = getNumber(value);

  if (temperature === null) return "#64748b";
  if (temperature <= -25) return "#4338ca";
  if (temperature <= -15) return "#2563eb";
  if (temperature <= -5) return "#0284c7";
  if (temperature < 5) return "#0891b2";
  if (temperature < 15) return "#16a34a";
  if (temperature < 25) return "#d97706";

  return "#dc2626";
}

function getSelectedWeatherDate(weather: CurrentWeather | null): string {
  if (!weather) return "Мэдээлэл алга";

  const value =
    weather.ajiglasan_ognoo ??
    weather.hemjsen_ognoo ??
    weather.hadgalsan_ognoo;

  if (!value) return "Огноо тодорхойгүй";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return String(value);
  }

  return new Intl.DateTimeFormat("mn-MN", {
    month: "2-digit",
    day: "2-digit",
    hour: "2-digit",
    minute: "2-digit",
    hour12: false,
    timeZone: "Asia/Ulaanbaatar",
  }).format(date);
}

function createTemperatureIcon(
  temperature: unknown,
  selected: boolean,
): L.DivIcon {
  const color = getTemperatureColor(temperature);
  const label = formatTemperature(temperature);

  return L.divIcon({
    className: "wx-marker-wrap",
    html: `
      <div
        class="wx-marker ${selected ? "is-selected" : ""}"
        style="--c:${color}"
      >
        <b>${label}</b>
      </div>
    `,
    iconSize: [0, 0],
    iconAnchor: [0, 0],
    popupAnchor: [0, -12],
  });
}

function MapSetup() {
  const map = useMap();

  useEffect(() => {
    map.setView(MONGOLIA_CENTER, 5);
    const timer = window.setTimeout(() => {
      map.invalidateSize();
    }, 150);

    return () => window.clearTimeout(timer);
  }, [map]);

  return null;
}

function MapResize() {
  const map = useMapEvents({
    zoomend: () => {
      map.invalidateSize();
    },
  });

  useEffect(() => {
    const timer = window.setTimeout(() => {
      map.invalidateSize();
    }, 300);

    return () => window.clearTimeout(timer);
  }, [map]);

  return null;
}

export default function MongoliaMap({
  geoJson,
  selectedId,
  onSelect,
}: MongoliaMapProps) {
  const [stations, setStations] = useState<StationWeather[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  const loadStations = useCallback(async () => {
    setLoading(true);
    setError("");

    try {
      const response = await fetch(
        `${API_BASE_URL}/weather_db/stantsuud`,
        { cache: "no-store" },
      );

      if (!response.ok) {
        throw new Error(`Станцын API алдаа: ${response.status}`);
      }

      const stationJson: unknown = await response.json();

      const normalizedStations: Station[] = getArray(stationJson)
        .map((item) => {
          if (!item || typeof item !== "object") return null;

          const row = item as Record<string, unknown>;

          const id = row.stants_dugaar ?? row.stantsDugaar ?? row.id;
          const latitude = getNumber(
            row.urgurug ?? row.latitude ?? row.lat,
          );
          const longitude = getNumber(
            row.urtrag ?? row.longitude ?? row.lng ?? row.lon,
          );

          if (
            id === null ||
            id === undefined ||
            latitude === null ||
            longitude === null
          ) {
            return null;
          }

          if (
            latitude < -90 ||
            latitude > 90 ||
            longitude < -180 ||
            longitude > 180
          ) {
            return null;
          }

          return {
            stants_dugaar: id as number | string,
            ner: String(row.ner ?? row.name ?? `Станц ${id}`),
            ner_en: row.ner_en ? String(row.ner_en) : undefined,
            aimag: row.aimag ? String(row.aimag) : undefined,
            urgurug: latitude,
            urtrag: longitude,
          };
        })
        .filter((item): item is Station => item !== null);

      if (normalizedStations.length === 0) {
        throw new Error(
          "Станцын жагсаалт хоосон байна. API-ийн хариуг шалгана уу.",
        );
      }

      // Станц бүрийн одоогийн цаг агаарыг авна.
      // Алдаа гарсан нэг станц бусад станцын газрын зурагт нөлөөлөхгүй.
      const results = await Promise.all(
        normalizedStations.map(async (station): Promise<StationWeather> => {
          try {
            const weatherResponse = await fetch(
              `${API_BASE_URL}/weather_db/tsag_agaar/current?station=${encodeURIComponent(
                String(station.stants_dugaar),
              )}`,
              { cache: "no-store" },
            );

            if (!weatherResponse.ok) {
              return { ...station, weather: null };
            }

            const weatherJson: unknown = await weatherResponse.json();

            return {
              ...station,
              weather: getWeatherObject(weatherJson),
            };
          } catch {
            return { ...station, weather: null };
          }
        }),
      );

      setStations(results);
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Газрын зургийн мэдээлэл авахад алдаа гарлаа.",
      );
      setStations([]);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    void loadStations();
  }, [loadStations]);

  const selectedStation = useMemo(() => {
    if (selectedId === null || selectedId === undefined) return null;

    return (
      stations.find(
        (station) =>
          String(station.stants_dugaar) === String(selectedId),
      ) ?? null
    );
  }, [stations, selectedId]);

  const handleSelect = useCallback(
    (station: StationWeather) => {
      onSelect?.(station.stants_dugaar);
    },
    [onSelect],
  );

  return (
    <div className="mongolia-map-wrapper">
      {loading && (
        <div className="map-status" role="status">
          Станцуудын цаг агаарын мэдээллийг ачаалж байна…
        </div>
      )}

      {error && (
        <div className="map-status map-status-error" role="alert">
          <p>{error}</p>
          <button type="button" onClick={() => void loadStations()}>
            Дахин оролдох
          </button>
        </div>
      )}

      <MapContainer
        center={MONGOLIA_CENTER}
        zoom={5}
        minZoom={3}
        maxZoom={12}
        scrollWheelZoom
        className="mongolia-map"
      >
        {/* Хиймэл дагуулын зураг */}
        <TileLayer
          attribution="Tiles © Esri"
          url="https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}"
        />

        {/* Монгол Улсын хил */}
        {geoJson && (
          <GeoJSON
            key="mongolia-boundary"
            data={geoJson}
            style={() => ({
              color: "#00bfff",
              weight: 3,
              opacity: 1,
              fillColor: "#168bff",
              fillOpacity: 0.16,
            })}
          />
        )}

        {/* Бүх станцын температур */}
        {stations.map((station) => {
          const selected =
            selectedId !== null &&
            selectedId !== undefined &&
            String(station.stants_dugaar) === String(selectedId);

          return (
            <Marker
              key={String(station.stants_dugaar)}
              position={[station.urgurug, station.urtrag]}
              icon={createTemperatureIcon(
                station.weather?.temperatur_c,
                selected,
              )}
              eventHandlers={{
                click: () => handleSelect(station),
              }}
            >
              <Popup className="wx-popup">
                <div className="wx-pop">
                  <div className="wx-pop-head">
                    <div>
                      <h3>{station.ner}</h3>
                      <span>{station.aimag || "Монгол Улс"}</span>
                    </div>
                  </div>

                  <div className="wx-pop-temp">
                    {formatTemperature(station.weather?.temperatur_c)}
                  </div>

                  <p className="wx-pop-cond">
                    {station.weather?.tsag_agaar_nohtsol ||
                      "Цаг агаарын нөхцөл тодорхойгүй"}
                  </p>

                  <dl className="wx-pop-stats">
                    <div>
                      <dt>Мэдрэгдэх</dt>
                      <dd>
                        {formatTemperature(station.weather?.medregdeh_c)}
                      </dd>
                    </div>

                    <div>
                      <dt>Салхи</dt>
                      <dd>
                        {getNumber(station.weather?.salhiny_hurd_ms) !== null
                          ? `${getNumber(
                              station.weather?.salhiny_hurd_ms,
                            )} м/с`
                          : "—"}
                      </dd>
                    </div>

                    <div>
                      <dt>Чийгшил</dt>
                      <dd>
                        {getNumber(station.weather?.chigleg_huvi) !== null
                          ? `${getNumber(
                              station.weather?.chigleg_huvi,
                            )}%`
                          : "—"}
                      </dd>
                    </div>

                    <div>
                      <dt>Даралт</dt>
                      <dd>
                        {getNumber(station.weather?.daralt_hpa) !== null
                          ? `${getNumber(
                              station.weather?.daralt_hpa,
                            )} hPa`
                          : "—"}
                      </dd>
                    </div>
                  </dl>

                  <p className="wx-pop-date">
                    Шинэчилсэн: {getSelectedWeatherDate(station.weather)}
                  </p>

                  <button
                    type="button"
                    className="wx-select-button"
                    onClick={() => handleSelect(station)}
                  >
                    Энэ станцыг сонгох
                  </button>
                </div>
              </Popup>
            </Marker>
          );
        })}

        <MapSetup />
        <MapResize />
      </MapContainer>

      {/* Өнгөний тайлбар */}
      <div className="map-legend">
        <strong>Температур</strong>
        <div className="map-legend-gradient" />
        <div className="map-legend-ticks">
          <span>-25°</span>
          <span>-15°</span>
          <span>-5°</span>
          <span>5°</span>
          <span>15°</span>
          <span>25°+</span>
        </div>
      </div>

      {selectedStation && (
        <div className="map-selected-label">
          <span className="map-selected-dot" />
          <div>
            <strong>{selectedStation.ner}</strong>
            <span>
              {formatTemperature(selectedStation.weather?.temperatur_c)}
              {" · "}
              {selectedStation.weather?.tsag_agaar_nohtsol || "Мэдээлэл алга"}
            </span>
          </div>
        </div>
      )}
    </div>
  );
}
